\olsection{\usetoken{S}{derivability} e Consistência}

Estabeleceremos agora uma série de propriedades da relação de
!!{derivability}.  Elas são interessantes por si mesmas, mas cada uma terá um
papel na prova do teorema da completude.

\begin{prop}\ollabel{prop:provability-contr}
  Se $\Gamma \Proves !A$ e $\Gamma \cup \{!A\}$ é
  inconsistente, então $\Gamma$ é inconsistente.
\end{prop}

\begin{proof}
Há $\Gamma_0$ e $\Gamma_1 \subseteq \Gamma$ finitos tais que
$\Log{LK}$ !!{derive} $\Gamma_0 \Sequent !A$ e $!A, \Gamma_1
\Sequent \quad$.  Seja $\pi_0$ a $\Log{LK}$-!!{derivation} de $\Gamma_0 \Sequent
!A$ e $\pi_1$ a $\Log{LK}$-!!{derivation} de $\Gamma_1, !A
\Sequent \quad$. Podemos então derivar
\begin{prooftree}
\AxiomC{}
\RightLabel{$\pi_0$}
\Deduce$ \Gamma_0 \fCenter !A $
\AxiomC{}
\RightLabel{$\pi_1$}
\Deduce$!A, \Gamma_1 \fCenter $
\RightLabel{\Cut}
\BinaryInf$ \Gamma_0,\Gamma_1 \fCenter $
\end{prooftree}

Como $\Gamma_0 \subseteq \Gamma$ e $\Gamma_1 \subseteq \Gamma$,
$\Gamma_0 \cup \Gamma_1 \subseteq \Gamma$; portanto, $\Gamma$~é inconsistente.
\end{proof}

\begin{prop}
\ollabel{prop:prov-incons}
$\Gamma \Proves !A$ se, e somente se, $\Gamma \cup \{\lnot !A\}$ é inconsistente.
\end{prop}

\begin{proof}
Primeiro, suponha $\Gamma \Proves !A$, isto é, há
!!a{derivation}~$\pi_0$ de $\Gamma \Sequent !A$. Acrescentando uma
regra $\LeftR{\lnot}$, obtemos !!a{derivation} de~$\lnot !A, \Gamma
\Sequent \quad$, isto é, $\Gamma \cup \{\lnot !A\}$ é inconsistente.

Se $\Gamma \cup \{\lnot !A\}$ é inconsistente, há
!!a{derivation}~$\pi_1$ de $\lnot !A, \Gamma \Sequent \quad$. O
seguinte é !!a{derivation} de $\Gamma \Sequent !A$:
\begin{prooftree}
  \Axiom$!A \fCenter !A$
  \RightLabel{\RightR{\lnot}}
  \UnaryInf$\fCenter !A, \lnot !A$
  \AxiomC{}
  \RightLabel{$\pi_1$}
  \Deduce$\lnot !A, \Gamma \fCenter$
  \RightLabel{\Cut}
  \BinaryInf$\Gamma \fCenter !A$
\end{prooftree}
\end{proof}

\begin{prob}
Prove que $\Gamma \Proves \lnot !A$ se, e somente se, $\Gamma \cup \{!A\}$ é inconsistente.
\end{prob}

\begin{prop}\ollabel{prop:explicit-inc}
  Se $\Gamma \Proves !A$ e $\lnot !A \in \Gamma$, então $\Gamma$ é
  inconsistente.
\end{prop}

\begin{proof}
  Suponha $\Gamma \Proves !A$ e $\lnot !A \in \Gamma$.  Então há
  !!a{derivation}~$\pi$ de um sequente $\Gamma_0 \Sequent !A$. O
  sequente $\lnot !A, \Gamma_0 \Sequent \quad$ também é !!{derivable}:
  \begin{prooftree}
    \AxiomC{}
    \RightLabel{$\pi$}
    \Deduce$\Gamma_0 \fCenter !A$
    \Axiom$!A \fCenter !A$
    \RightLabel{\LeftR{\lnot}}
    \UnaryInf$\lnot !A, !A \fCenter$
    \RightLabel{\LeftR{\Exchange}}
    \UnaryInf$!A, \lnot !A \fCenter$
    \RightLabel{\Cut}
    \BinaryInf$\Gamma_0, \lnot !A \fCenter$
  \end{prooftree}
  Como $\lnot !A \in \Gamma$ e $\Gamma_0 \subseteq \Gamma$, isso
  mostra que $\Gamma$ é inconsistente.
\end{proof}

\begin{prop}\ollabel{prop:provability-exhaustive}
  Se $\Gamma \cup \{!A\}$ e $\Gamma \cup \{\lnot !A\}$ são ambos
  inconsistentes, então $\Gamma$ é inconsistente.
\end{prop}

\begin{proof}
Há conjuntos finitos $\Gamma_0 \subseteq \Gamma$ e $\Gamma_1
\subseteq \Gamma$ e $\Log{LK}$-!!{derivation}s $\pi_0$ e $\pi_1$
de $!A, \Gamma_0 \Sequent \quad$ e $\lnot !A, \Gamma_1 \Sequent
\quad$, respectivamente. Podemos então derivar
\begin{prooftree}
\AxiomC{}
\RightLabel{$\pi_0$}
\Deduce$ !A, \Gamma_0 \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$ \Gamma_0 \fCenter \lnot !A$
\AxiomC{}
\RightLabel{$\pi_1$}
\Deduce$\lnot !A, \Gamma_1 \fCenter  $
\RightLabel{\Cut}
\BinaryInf$ \Gamma_0, \Gamma_1 \fCenter $
\end{prooftree}
Como $\Gamma_0 \subseteq \Gamma$ e $\Gamma_1 \subseteq \Gamma$,
$\Gamma_0 \cup \Gamma_1 \subseteq \Gamma$. Portanto, $\Gamma$ é
inconsistente.
\end{proof}
