\olsection{\usetoken{S}{derivability} e Consistência}

Estabeleceremos agora várias propriedades da relação de
!!{derivability}.  Elas são independentemente interessantes, mas cada
uma desempenhará um papel na prova do teorema da completude.

\begin{prop}\ollabel{prop:provability-contr}
  Se $\Gamma \Proves !A$ e $\Gamma \cup \{!A\}$ é inconsistente,
  então $\Gamma$ é inconsistente.
\end{prop}

\begin{proof}
Seja a !!{derivation} de~$!A$ a partir de~$\Gamma$ igual a~$\delta_1$ e a
!!{derivation} de~$\lfalse$ a partir de $\Gamma \cup \{!A\}$
igual a~$\delta_2$. Podemos então derivar:
\begin{prooftree}
\AxiomC{$\Gamma, \Discharge{!A}{1}$}
\RightLabel{$\delta_2$}
\DeduceC{$\lfalse$}
\DischargeRule{\Intro{\lnot}}{1}
\UnaryInfC{$\lnot !A$}
\AxiomC{$\Gamma$}
\RightLabel{$\delta_1$}
\DeduceC{$!A$}
\RightLabel{\Elim{\lnot}}
\BinaryInfC{$\lfalse$}
\end{prooftree}
Na nova !!{derivation}, a hipótese~$!A$ é !!{discharged}, de modo que é
uma !!{derivation} a partir de~$\Gamma$.
\end{proof}

\begin{prop}
\ollabel{prop:prov-incons}
$\Gamma \Proves !A$ se, e somente se, $\Gamma \cup \{\lnot !A\}$ é inconsistente.
\end{prop}

\begin{proof}
Primeiro suponha que $\Gamma \Proves !A$, isto é, há uma
!!{derivation}~$\delta_0$ de~$!A$ a partir de hipóteses
!!{undischarged}s~$\Gamma$. Obtemos uma !!{derivation} de $\lfalse$ a partir de
$\Gamma \cup \{\lnot !A\}$ da seguinte forma:
\begin{prooftree}
  \AxiomC{$\lnot !A$}
  \AxiomC{$\Gamma$}
  \RightLabel{$\delta_0$}
  \DeduceC{$!A$}
  \RightLabel{\Elim{\lnot}}
  \BinaryInfC{$\lfalse$}
\end{prooftree}

Agora suponha que $\Gamma \cup \{\lnot !A\}$ é inconsistente, e seja
$\delta_1$ a !!{derivation} correspondente de~$\lfalse$ a partir de
hipóteses !!{undischarged}s em~$\Gamma \cup \{\lnot !A\}$. Obtemos
uma !!{derivation} de~$!A$ apenas a partir de~$\Gamma$ usando~\FalseCl:
\begin{prooftree}
  \AxiomC{$\Gamma, \Discharge{\lnot !A}{1}$}
  \RightLabel{$\delta_1$}
  \DeduceC{$\lfalse$}
  \RightLabel{\FalseCl}
  \DischargeRule{\FalseCl}{1}
  \UnaryInfC{$!A$}
\end{prooftree}
\end{proof}

\begin{prob}
Prove que $\Gamma \Proves \lnot !A$ se, e somente se, $\Gamma \cup \{!A\}$ é
inconsistente.
\end{prob}

\begin{prop}\ollabel{prop:explicit-inc}
  Se $\Gamma \Proves !A$ e $\lnot !A \in \Gamma$, então $\Gamma$ é
  inconsistente.
\end{prop}

\begin{proof}
  Suponha que $\Gamma \Proves !A$ e $\lnot !A \in \Gamma$.  Então há
  uma !!{derivation}~$\delta$ de~$!A$ a partir de~$\Gamma$. Considere esta
    aplicação simples da regra $\Elim{\lnot}$:
  \begin{prooftree}
    \AxiomC{$\lnot !A$}
    \AxiomC{$\Gamma$}
    \RightLabel{$\delta$}
    \DeduceC{$!A$}
    \RightLabel{\Elim{\lnot}}
    \BinaryInfC{$\lfalse$}
  \end{prooftree}
  Como $\lnot !A \in \Gamma$, todas as hipóteses !!{undischarged}s estão
  em~$\Gamma$; isto mostra que $\Gamma \Proves \lfalse$.
\end{proof}

\begin{prop}\ollabel{prop:provability-exhaustive}
  Se $\Gamma \cup \{!A\}$ e $\Gamma \cup \{\lnot !A\}$ são ambos
  inconsistentes, então $\Gamma$ é inconsistente.
\end{prop}

\begin{proof}
Há !!{derivation}s $\delta_1$ e $\delta_2$ de~$\lfalse$ a partir de
  $\Gamma \cup \{ !A \}$ e de $\lfalse$ a partir de $\Gamma \cup \{ \lnot !A
  \}$, respectivamente. Podemos então derivar
\begin{prooftree}
\AxiomC{$\Gamma, \Discharge{\lnot !A}{2}$}
\RightLabel{$\delta_2$}
\DeduceC{$\lfalse$}
\DischargeRule{\Intro{\lnot}}{2}
\UnaryInfC{$\lnot \lnot !A$}
\AxiomC{$\Gamma, \Discharge{!A}{1}$}
\RightLabel{$\delta_1$}
\DeduceC{$\lfalse$}
\DischargeRule{\Intro{\lnot}}{1}
\UnaryInfC{$\lnot !A$}
\RightLabel{\Elim{\lnot}}
\BinaryInfC{$\lfalse$}
\end{prooftree}
Como as hipóteses $!A$ e $\lnot !A$ são !!{discharged}s, isto é
uma !!{derivation} de~$\lfalse$ apenas a partir de~$\Gamma$. Logo $\Gamma$ é
inconsistente.
\end{proof}
