\olsection{\usetoken{S}{derivability} e os Quantificadores}

\begin{explain}
  O teorema da completude também requer que as regras de dedução
  natural produzam os fatos sobre~$\Proves$ estabelecidos nesta seção.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} Se $c$ é uma constante que não
ocorre em $\Gamma$ ou $!A(x)$ e $\Gamma \Proves !A(c)$, então $\Gamma
\Proves \lforall[x][!A(x)]$.
\end{thm}

\begin{proof}
Seja $\delta$ uma !!{derivation} de $!A(c)$ a partir de $\Gamma$.
Adicionando uma inferência \Intro{\lforall}, obtemos uma !!{derivation}
de $\lforall[x][!A(x)]$. Como $c$ não ocorre em $\Gamma$ ou $!A(x)$,
a condição de variável própria é satisfeita.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}

\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{A seguinte é uma !!{derivation} de
  $\lexists[x][!A(x)]$ a partir de~$!A(t)$:
\begin{prooftree}
\AxiomC{$!A(t)$}
\RightLabel{\Intro{\lexists}}
\UnaryInfC{$\lexists[x][!A(x)]$}
\end{prooftree}}{}

\tagitem{prvAll}{A seguinte é uma !!{derivation} de~$!A(t)$
  a partir de~$\lforall[x][!A(x)]$:
\begin{prooftree}
\AxiomC{$\lforall[x][!A(x)]$}
\RightLabel{\Elim{\lforall}}
\UnaryInfC{$!A(t)$}
\end{prooftree}}{}
\end{tagenumerate}
\end{proof}
