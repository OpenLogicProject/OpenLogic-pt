\olsection{Uma Prova Direta do Teorema da Compacidade}

Podemos provar o Teorema da Compacidade diretamente, sem apelar ao
Teorema da Completude, usando as mesmas ideias da prova do
teorema da completude.  Na prova do Teorema da Completude,
começamos com um conjunto~$\Gamma$ consistente de !!{sentence}s,
expandimos-o para um conjunto~$\Gamma^*$ de !!{sentence}s consistente\iftag{FOL}{, saturado,}{} e
!!{complete}, e então mostramos que no
\iftag{FOL}{modelo de termos~$\Struct{M(\Gamma^*)}$}{!!{valuation}~$\pAssign{v(\Gamma^*)}$}
construído a partir de $\Gamma^*$, todas as !!{sentence}s de~$\Gamma$ são verdadeiras,
de modo que $\Gamma$ é satisfatível.

Podemos usar o mesmo método para mostrar que um conjunto finitamente
satisfatível de sentenças é satisfatível. Só temos que provar as
versões correspondentes dos resultados que levam ao lema da verdade,
substituindo ``consistente'' por ``finitamente satisfatível''.

\begin{prop}
\ollabel{prop:fsat-ccs}
Suponha que $\Gamma$ é !!{complete} e finitamente satisfatível. Então:
\begin{enumerate}
\tagitem{prvAnd}{$(!A \land !B) \in \Gamma$
  se, e somente se, tanto $!A \in \Gamma$ quanto $!B \in \Gamma$.}{}

\tagitem{prvOr}{$(!A \lor !B) \in \Gamma$ se, e somente se,
  $!A \in \Gamma$ ou $!B \in \Gamma$.}{}

\tagitem{prvIf}{$(!A \lif !B) \in \Gamma$ se, e somente se,
  $!A \notin \Gamma$ ou $!B \in \Gamma$.}{}
\end{enumerate}
\end{prop}

\tagprob{FOL}
\begin{prob}
Prove \olref[fol][com][cpd]{prop:fsat-ccs}. Evite o uso de $\Proves$.
\end{prob}
\tagendprob

\tagprob{notFOL}
\begin{prob}
Prove \olref[pl][com][cpd]{prop:fsat-ccs}. Evite o uso de $\Proves$.
\end{prob}
\tagendprob

\iftag{FOL}{%
\begin{lem}
\ollabel{lem:fsat-henkin} Todo conjunto finitamente satisfatível~$\Gamma$ pode
ser estendido a um conjunto saturado finitamente satisfatível~$\Gamma'$.
\end{lem}
}{}

\tagprob{FOL}
\begin{prob}
Prove \olref[fol][com][cpd]{lem:fsat-henkin}. (Dica: o passo crucial
é mostrar que se $\Gamma_n$ é finitamente satisfatível, então $\Gamma_n
\cup \{!D_n\}$ também o é, sem qualquer apelo a !!{derivation}s ou consistência.)
\end{prob}
\tagendprob

\iftag{FOL}{
\begin{prop}\ollabel{prop:fsat-instances}
Suponha que $\Gamma$ é completo, finitamente satisfatível e saturado.
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$\lexists[x][!A(x)] \in \Gamma$ se, e somente se, $!A(t) \in \Gamma$
  para pelo menos um termo fechado~$t$.}{}
\tagitem{prvAll}{$\lforall[x][!A(x)] \in \Gamma$ se, e somente se, $!A(t) \in \Gamma$
  para todos os termos fechados~$t$.}{}
\end{tagenumerate}
\end{prop}
}{}

\tagprob{FOL}
\begin{prob}
Prove \olref[fol][com][cpd]{prop:fsat-instances}.
\end{prob}
\tagendprob

\begin{lem}
\ollabel{lem:fsat-lindenbaum} Todo conjunto finitamente satisfatível~$\Gamma$
pode ser estendido a um conjunto !!{complete} e finitamente satisfatível
~$\Gamma^*$.
\end{lem}

\tagprob{FOL}
\begin{prob}
Prove \olref[fol][com][cpd]{lem:fsat-lindenbaum}.  (Dica: o passo crucial
é mostrar que se $\Gamma_n$ é finitamente satisfatível, então
ou $\Gamma_n \cup \{!A_n\}$ ou $\Gamma_n \cup \{\lnot !A_n\}$ é
finitamente satisfatível.)
\end{prob}
\tagendprob

\tagprob{notFOL}
\begin{prob}
Prove \olref[pl][com][cpd]{lem:fsat-lindenbaum}.  (Dica: o passo crucial
é mostrar que se $\Gamma_n$ é finitamente satisfatível, então
ou $\Gamma_n \cup \{!A_n\}$ ou $\Gamma_n \cup \{\lnot !A_n\}$ é
finitamente satisfatível.)
\end{prob}
\tagendprob

\begin{thm}[Compacidade]
\ollabel{thm:compactness-direct} $\Gamma$ é satisfatível se, e somente se,
é finitamente satisfatível.
\end{thm}

\begin{proof}
Se $\Gamma$ é satisfatível, então há
\iftag{FOL}{!!a{structure}~$\Struct{M}$}{!!a{valuation}~$\pAssign{v}$}
tal que $\iftag{FOL}{\Sat{M}}{\pSat{v}}{!A}$ para todo $!A \in \Gamma$.
É claro que esta \iftag{FOL}{$\Struct{M}$}{$\pAssign{v}$} também
satisfaz todo subconjunto finito de~$\Gamma$, de modo que $\Gamma$ é finitamente
satisfatível.

Agora suponha que $\Gamma$ é finitamente satisfatível.\iftag{FOL}{ Por
  \olref{lem:fsat-henkin}, há um conjunto finitamente satisfatível e saturado
  $\Gamma' \supseteq \Gamma$.}{} Por \olref{lem:fsat-lindenbaum},
$\iftag{FOL}{\Gamma'}{\Gamma}$ pode ser estendido a um conjunto !!{complete} e
finitamente satisfatível~$\Gamma^*$\iftag{FOL}{, e $\Gamma^*$ ainda é
  saturado}{}. Construa o \iftag{FOL}{modelo de termos
  ~$\Struct{M(\Gamma^*)}$}{!!{valuation}~$\pAssign{v(\Gamma^*)}$}
como em \olref[mod]{defn:termmodel}.\iftag{FOL}{ Note que
\olref[mod]{prop:quant-termmodel} não dependeu do fato de que
$\Gamma^*$ é consistente (ou !!{complete} ou saturado, aliás), mas apenas do fato de que $\Struct{M(\Gamma^*)}$ é coberta.}{}
A prova do Lema da Verdade (\olref[mod]{lem:truth}) funciona se
substituirmos referências a \olref[ccs]{prop:ccs}\iftag{FOL}{ e
\olref[hen]{prop:saturated-instances} por referências a
\olref{prop:fsat-ccs} e \olref{prop:fsat-instances}}.
\end{proof}

\tagprob{FOL}
\begin{prob}
Escreva a prova completa do Lema da Verdade
(\olref[fol][com][mod]{lem:truth}) na versão necessária para a
prova de \olref[fol][com][cpd]{thm:compactness-direct}.
\end{prob}
\tagendprob

\tagprob{notFOL}
\begin{prob}
Escreva a prova completa do Lema da Verdade
(\olref[pl][com][mod]{lem:truth}) na versão necessária para a
prova de \olref[pl][com][cpd]{thm:compactness-direct}.
\end{prob}
\tagendprob
