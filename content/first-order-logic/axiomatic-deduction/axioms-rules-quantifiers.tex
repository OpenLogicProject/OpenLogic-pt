\olsection{Axiomas e regras para quantificadores}

\begin{defn}[Axiomas para quantificadores]
Os \emph{axiomas} que regem os quantificadores são
todas as instâncias dos seguintes:
\begin{align}
\ollabel{ax:q1} & \lforall[x][!B] \lif !B(t), \\
\ollabel{ax:q2} & !B(t) \lif \lexists[x][!B].
\end{align}
para qualquer termo fechado~$t$.
\end{defn}

\begin{defn}[Regras para quantificadores]
\item Se $!B \lif !A(a)$ já ocorre na !!{derivation} e $a$
  não ocorre em~$\Gamma$ nem em~$!B$, então $!B \lif
  \lforall[x][!A(x)]$ é um passo de inferência correto.
\item Se $!A(a) \lif !B$ já ocorre na !!{derivation} e $a$
  não ocorre em~$\Gamma$ nem em~$!B$, então $\lexists[x][!A(x)] \lif
  !B$ é um passo de inferência correto.
\end{defn}

Abreviaremos qualquer uma dessas regras por ``\QR''.
