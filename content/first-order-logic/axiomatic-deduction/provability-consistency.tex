\olsection{\usetoken{S}{derivability} e consistência}

Estabeleceremos agora algumas propriedades da relação de
!!{derivability}. Elas são interessantes por si mesmas, e cada uma
desempenhará um papel na demonstração do teorema da completude.

\begin{prop}\ollabel{prop:provability-contr}
  Se $\Gamma \Proves !A$ e $\Gamma \cup \{!A\}$ é inconsistente,
  então $\Gamma$ é inconsistente.
\end{prop}

\begin{proof}
  Se $\Gamma \cup \{!A\}$ é inconsistente, então $\Gamma \cup \{!A\}
  \Proves \lfalse$. Por \olref[ptn]{prop:reflexivity}, $\Gamma \Proves
  !B$ para todo $!B \in \Gamma$. Como também $\Gamma \Proves !A$ por
  hipótese, $\Gamma \Proves !B$ para todo $!B \in \Gamma \cup
  \{!A\}$. Por \olref[ptn]{prop:transitivity}, $\Gamma \Proves
  \lfalse$, O.s.\@, $\Gamma$ é inconsistente.
\end{proof}

\begin{prop}
\ollabel{prop:prov-incons}
$\Gamma \Proves !A$ se, e somente se, $\Gamma \cup \{\lnot !A\}$ é inconsistente.
\end{prop}

\begin{proof}
Primeiro, suponha $\Gamma \Proves !A$. Então $\Gamma \cup \{\lnot !A\}
\Proves !A$ por \olref[ptn]{prop:monotonicity}. Temos também
$\Gamma \cup \{\lnot !A\} \Proves \lnot !A$ por
\olref[ptn]{prop:reflexivity}. Também temos
$\Proves \lnot !A \lif (!A \lif \lfalse)$ por \olref[prp]{ax:lnot2}.
Portanto, por duas aplicações de \olref[ded]{prop:mp}, temos $\Gamma
\cup \{\lnot !A\} \Proves \lfalse$.

Agora suponha que $\Gamma \cup \{\lnot !A\}$ seja inconsistente, O.s.\@,
$\Gamma \cup \{\lnot !A\} \Proves \lfalse$. Pelo teorema da dedução,
$\Gamma \Proves \lnot !A \lif \lfalse$. Temos $\Gamma \Proves (\lnot !A
\lif \lfalse) \lif \lnot\lnot !A$ por \olref[prp]{ax:lfalse2}; logo,
$\Gamma \Proves \lnot\lnot !A$ por \olref[ded]{prop:mp}. Como $\Gamma
\Proves \lnot\lnot !A \lif !A$ (\olref[prp]{ax:dne}), temos novamente
$\Gamma \Proves !A$ por \olref[ded]{prop:mp}.
\end{proof}

\begin{prob}
Demonstre que $\Gamma \Proves \lnot !A$ se, e somente se,
$\Gamma \cup \{!A\}$ é inconsistente.
\end{prob}

\begin{prop}\ollabel{prop:explicit-inc}
  Se $\Gamma \Proves !A$ e $\lnot !A \in \Gamma$, então $\Gamma$ é
  inconsistente.
\end{prop}

\begin{proof}
  $\Gamma \Proves \lnot !A \lif (!A \lif \lfalse)$ por \olref[prp]{ax:lnot2}.
  $\Gamma \Proves \lfalse$ por duas aplicações de \olref[ded]{prop:mp}.
\end{proof}


\begin{prop}\ollabel{prop:provability-exhaustive}
  Se $\Gamma \cup \{!A\}$ e $\Gamma \cup \{\lnot !A\}$ são ambos
  inconsistentes, então $\Gamma$ é inconsistente.
\end{prop}

\begin{proof}
Exercício.
\end{proof}
