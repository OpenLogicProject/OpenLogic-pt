\olsection{\usetoken{S}{derivability} e os Quantificadores}

\begin{explain}
  O teorema da completude também exige que as deduções axiomáticas
  forneçam os fatos sobre~$\Proves$ estabelecidos nesta seção.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} Se $c$ for !!a{constant} que não ocorre
em $\Gamma$ nem em $!A(x)$ e $\Gamma \Proves !A(c)$, então $\Gamma
\Proves \lforall[x][!A(x)]$.
\end{thm}

\begin{proof}
Pelo teorema da dedução, $\Gamma \Proves \ltrue \lif !A(c)$. Como
$c$ não ocorre em $\Gamma$ nem em~$\top$, obtemos $\Gamma \Proves
\ltrue \lif \lforall[x][!A(x)]$. Aplicando novamente o teorema da dedução, $\Gamma \Proves
\lforall[x][!A(x)]$.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}

\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{Por \olref[qua]{ax:q2} e pelo teorema da dedução.}{}
\tagitem{prvAll}{Por \olref[qua]{ax:q1} e pelo teorema da dedução.}{}
\end{tagenumerate}
\end{proof}
