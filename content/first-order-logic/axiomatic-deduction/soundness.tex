\olsection{Correção}

\begin{explain}
Um sistema com !!^a{derivation}, como a dedução axiomática, é \emph{correto}
se não pode !!{derive} coisas que de fato não valem. A correção é,
portanto, uma espécie de propriedade de segurança garantida para
sistemas de !!{derivation}. Dependendo da propriedade da teoria da
prova em questão, gostaríamos de saber, por exemplo, que
\begin{enumerate}
\item todo !!{derivable}~$!A$ é válido;
\item se $!A$ é !!{derivable} a partir de algumas outras
  fórmulas~$\Gamma$, também é consequência delas;
\item se um conjunto de !!{formula}s~$\Gamma$ é inconsistente, ele é
  insatisfatível.
\end{enumerate}
Essas são propriedades importantes de um sistema com !!a{derivation}. Se
alguma delas não vale, o sistema de !!{derivation} é deficiente---ele
!!{derive} em excesso. Consequentemente, estabelecer a correção de
um sistema com !!a{derivation} é de suma importância.
\end{explain}

\begin{prop}
  Se $!A$ é um axioma, então
  \iftag{FOL}
    {$\Sat{M}{!A}[s]$ para cada !!{structure}~$\Struct{M}$ e atribuição~$s$.}
    {$\pSat{v}{!A}$ para cada !!{valuation}~$\pAssign{v}$.}
\end{prop}

\begin{proof}
\iftag{FOL}{Temos de verificar que todos os axiomas são válidos. Por
  exemplo, eis o caso de \olref[qua]{ax:q1}: suponha que $t$ seja
  !!{free for} $x$ em $!A$ e assuma
  $\Sat{M}{\lforall[x][!A]}[s]$. Então, pela definição de satisfação,
  para cada $\varAssign{s'}{s}{x}$, também $\Sat{M}{!A}[s']$ e, em
  particular, isso vale quando $s'(x) = \Value{t}{M}[s]$. Por
  \olref[syn][ext]{prop:ext-formulas},
  $\Sat{M}{\Subst{!A}{t}{x}}[s]$. Isso mostra que
  $\Sat{M}{(\lforall[x][!A] \lif \Subst{!A}{t}{x})}[s]$.}{Faça
  tabelas de verdade para cada axioma a fim de verificar que são
  tautologias.}
\end{proof}

\begin{thm}[Correção]
\ollabel{thm:soundness}
Se $\Gamma \Proves !A$, então $\Gamma \Entails !A$.
\end{thm}

\begin{proof}
Por indução no comprimento da !!{derivation} de $!A$ a partir de
$\Gamma$. Se não há passos justificados por inferências, então todas
as !!{formula}s na !!{derivation} são instâncias de axiomas ou estão
em~$\Gamma$. Pela proposição anterior, todos os axiomas são
\iftag{FOL}{válidos}{tautologias}; portanto, se $!A$ é um axioma,
$\Gamma \Entails !A$. Se $!A \in \Gamma$, então, trivialmente,
$\Gamma \Entails !A$.

Se o último passo da !!{derivation} de~$!A$ é justificado por modus
ponens, então há !!{formula}s $!B$ e $!B \lif !A$ na !!{derivation}, e
a hipótese de indução se aplica à parte da !!{derivation} que termina
nessas !!{formula}s (pois essa parte contém pelo menos um passo a menos
justificado por uma inferência). Assim, pela hipótese de indução,
$\Gamma \Entails !B$ e $\Gamma \Entails !B \lif !A$. Então, $\Gamma
\Entails !A$ por
\iftag{FOL}
      {\olref[syn][sem]{thm:sem-deduction}}
      {\olref[pl][syn][sem]{thm:sem-deduction}}.

\iftag{FOL}{Suponha agora que o último passo seja justificado por
\QR. Então esse passo tem a forma $!C \lif \lforall[x][B(x)]$ e há um
passo anterior $!C \lif !B(c)$ no qual $c$ não está em $\Gamma$, em
$!C$ nem em $\lforall[x][B(x)]$. Pela hipótese de indução, $\Gamma
\Entails !C \lif !B(c)$. Por
\olref[syn][sem]{thm:sem-deduction}, $\Gamma \cup \{!C\} \Entails
!B(c)$.

Considere alguma estrutura~$\Struct{M}$ tal que $\Sat{M}{\Gamma \cup
  \{!C\}}$. Precisamos mostrar que $\Sat{M}{\lforall[x][!B(x)]}$. Como
$\lforall[x][!B(x)]$ é !!a{sentence}, isso significa que temos de
mostrar, para toda atribuição de variável~$s$, que
$\Sat{M}{!B(x)}[s]$ (\olref[syn][ass]{prop:sat-quant}). Como $\Gamma
\cup \{!C\}$ consiste inteiramente em sentenças, $\Sat{M}{!D}[s]$
para todo $!D \in \Gamma$ por \olref[syn][sat]{defn:satisfaction}.
Seja $\Struct{M'}$ como $\Struct{M}$, exceto que $\Assign{c}{M'} =
s(x)$. Como $c$ não ocorre em~$\Gamma$ nem em~$!C$,
$\Sat{M'}{\Gamma \cup \{!C\}}$ por
\olref[syn][ext]{cor:extensionality-sent}. Como $\Gamma \cup \{!C\}
\Entails !B(c)$, $\Sat{M'}{B(c)}$. Como $!B(c)$ é !!a{sentence},
$\Sat{M'}{!B(c)}[s]$ por
\olref[syn][ass]{prop:sentence-sat-true}. $\Sat{M'}{!B(x)}[s]$ sse
$\Sat{M'}{!B(c)}[s]$ por \olref[syn][ext]{prop:ext-formulas} (recorde
que $!B(c)$ é simplesmente $\Subst{!B(x)}{c}{x}$). Logo,
$\Sat{M'}{!B(x)}[s]$. Como $c$ não ocorre em~$!B(x)$, por
\olref[syn][ext]{prop:extensionality}, $\Sat{M}{!B(x)}[s]$. Mas $s$
era uma atribuição de variável arbitrária; logo,
$\Sat{M}{\lforall[x][!B(x)]}$. Assim, $\Gamma \cup \{!C\} \Entails
\lforall[x][!B(x)]$. Por \olref[syn][sem]{thm:sem-deduction}, $\Gamma
\Entails !C \lif \lforall[x][!B(x)]$.

O caso em que $!A$ é justificado por \QR{}, mas tem a forma
$\lexists[x][!B(x)] \lif !C$, é deixado como exercício.}{}
\end{proof}

\tagprob{FOL}
\begin{prob}
  Complete a demonstração de \olref[fol][axd][sou]{thm:soundness}.
\end{prob}
\tagendprob

\begin{cor}
\ollabel{cor:weak-soundness}
Se $\Proves !A$, então $!A$ é \iftag{FOL}{válido}{uma tautologia}.
\end{cor}

\begin{cor}
\ollabel{cor:consistency-soundness}
Se $\Gamma$ é satisfatível, então é consistente.
\end{cor}

\begin{proof}
Demonstramos a contrapositiva. Suponha que $\Gamma$ não seja
consistente. Então $\Gamma \Proves \lfalse$, isto é, existe
!!a{derivation} de $\lfalse$ a partir de~$\Gamma$. Por
\olref{thm:soundness}, qualquer
\iftag{FOL}{!!{structure}~$\Struct{M}$}{!!{valuation}~$\pAssign{v}$}
que satisfaça $\Gamma$ deve satisfazer~$\lfalse$. Como
\iftag{FOL}{$\Sat/{M}{\lfalse}$ para toda
  !!{structure}~$\Struct{M}$}{$\pSat/{v}{\lfalse}$ para toda
  !!{valuation}~$\pAssign{v}$}, nenhuma
\iftag{FOL}{$\Struct{M}$}{$\pAssign{v}$} pode satisfazer $\Gamma$, isto
é, $\Gamma$ não é satisfatível.
\end{proof}
