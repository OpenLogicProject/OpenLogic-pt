\olsection{\usetoken{S}{derivability} e os conectivos proposicionais}

\begin{explain}
  Estabelecemos que a relação de !!{derivability}~$\Proves$ da dedução
  axiomática é forte o bastante para estabelecer alguns fatos básicos
  que envolvem os conectivos proposicionais, como $!A \land !B \Proves
  !A$ e $!A, !A \lif !B \Proves !B$ (modus ponens). Esses fatos são
  necessários para a demonstração do teorema da completude.
\end{explain}

\begin{prop}\ollabel{prop:provability-land}
  \begin{enumerate}
  \item \ollabel{prop:provability-land-left} Tanto $!A \land !B \Proves
    !A$ quanto $!A \land !B \Proves !B$
  \item \ollabel{prop:provability-land-right} $!A, !B \Proves !A \land !B$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item De \olref[prp]{ax:land1} e \olref[prp]{ax:land1}, por
      modus ponens.
  \item De \olref[prp]{ax:land3}, por duas aplicações de
    modus ponens.
  \end{enumerate}
\end{proof}


\begin{prop}\ollabel{prop:provability-lor}
  \begin{enumerate}
  \item $!A \lor !B, \lnot !A, \lnot !B$ é inconsistente.
  \item Tanto $!A \Proves !A \lor !B$ quanto $!B \Proves !A \lor !B$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item De \olref[prp]{ax:lnot1}, obtemos $\Proves \lnot !A \lif (!A
    \lif \lfalse)$ e $\Proves \lnot !B \lif (!B \lif \lfalse)$. Assim,
    pelo teorema da dedução, temos $\{\lnot !A\} \Proves !A \lif
    \lfalse$ e $\{\lnot !B\} \Proves !B \lif \lfalse$. De
    \olref[prp]{ax:lor3}, obtemos $\{\lnot !A, \lnot !B\} \Proves (!A
    \lor !B) \lif \lfalse$. Pelo teorema da dedução, $\{!A \lor !B,
    \lnot !A, \lnot !B\} \Proves \lfalse$.
  \item De \olref[prp]{ax:lor1} e \olref[prp]{ax:lor2}, por modus
    ponens.
  \end{enumerate}
\end{proof}

\begin{prop}\ollabel{prop:provability-lif}
  \begin{enumerate}
  \item \ollabel{prop:provability-lif-left}  $!A, !A \lif !B \Proves !B$.
  \item \ollabel{prop:provability-lif-right}
    Tanto $\lnot !A \Proves !A \lif !B$ quanto $!B \Proves !A \lif !B$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item Podemos !!{derive}:
    \begin{derivation}
      1. & $!A$ & \Hyp\\
      2. & $!A \lif !B$ & \Hyp\\
      3. & $!B$ & 1, 2, \MP
    \end{derivation}
    
  \item Por \olref[prp]{ax:lnot2}, \olref[prp]{ax:lif1} e o teorema da
    dedução, respectivamente.
  \end{enumerate}
\end{proof}
