\olsection{Como mostrar que sistemas são distintos}

Em \olref[prs]{sec}, vimos como demonstrar que dois sistemas de lógica
modal são, na verdade, o mesmo sistema. \olref[snd]{thm:soundness} nos
permite mostrar que dois sistemas modais $\Sigma$ e $\Sigma'$ são
distintos encontrando !!a{formula}~$!A$ tal que $\Sigma' \Proves !A$,
mas que falha em um modelo de~$\Sigma$.

\begin{prop}
  $\Log{KD} \subsetneq \Log{KT}$
\end{prop}

\begin{proof} Esta é a contraparte sintática do fato semântico de que
  toda relação reflexiva é serial. Para mostrar $\Log{KD} \subseteq
  \Log{KT}$, precisamos ver que $\Log{KD} \Proves !B$ implica
  $\Log{KT} \Proves !B$, o que decorre de $\Log{KT} \Proves
  \Log{D}$, como mostrado em
  \olref[prs]{prop:S5facts}\olref[prs]{prop:S5facts-KT-D}. Para mostrar
  que a inclusão é própria, pela correção
  (\olref[snd]{thm:soundness}), basta exibir um modelo de \Log{KD} no
  qual \Ax{T}, isto é, $\Box p \lif p$, falha (uma tarefa fácil deixada
  como exercício), pois então, pela correção, $\Log{KD}
  \Proves/ \Box p \lif p$.
\end{proof}

\begin{prop}
  $\Log{KB} \neq \Log{K4}$. 
\end{prop}

\begin{proof}
  Construímos um modelo simétrico no qual alguma instância de \Ax{4}
  falha; como a instância é obviamente !!{derivable} para \Log{K4},
  mas não em \Log{KB}, seguirá que $\Log{K4} \nsubseteq \Log{KB}$.
  Considere o modelo simétrico $\mModel{M}$ de \olref{fig:Bnot4}. Como
  o modelo é simétrico, \Ax{K} e~\Ax{B} são verdadeiros em~$\mModel{M}$
  (por \olref[syn][sch]{prop:Kvalid} e
  \olref[frd][acc]{thm:soundschemas}, respectivamente). Contudo,
  $\mSat/{M}{\Box p \lif \Box\Box p}[w_1]$.
\end{proof}

\begin{thm}\ollabel{thm:KTBnot45}
  $\Log{KTB} \Proves/ \Log{4}$ e $\Log{KTB} \Proves/ \Log{5}$.
\end{thm}

\begin{proof}
  Por \olref[frd][acc]{thm:soundschemas}, sabemos que todas as
  instâncias de \Ax{T} e \Ax{B} são verdadeiras em todo modelo
  reflexivo e simétrico, respectivamente. Portanto, pela correção,
  basta encontrar um modelo reflexivo e simétrico que contenha um
  mundo no qual alguma instância de~\Ax{4} falhe, e fazer o mesmo
  para~\Ax{5}. Usamos o mesmo modelo para ambas as afirmações.
  Considere o modelo simétrico e reflexivo em \olref{fig:KTBnot45}.
  Então $\mSat/{M}{\Box p \to \Box \Box p}[w_1]$, logo \Ax{4} falha em
  $w_1$. De modo semelhante, $\mSat/{M}{\Diamond \lnot p \lif \Box
    \Diamond \lnot p}[w_2]$, logo a instância de~\Ax{5} com
  $!A = \lnot p$ falha em~$w_2$.
\end{proof}

\begin{thm}\ollabel{thm:KD5not4}
  $\Log{KD5} \neq \Log{KT4} = \Log{S4}$.
\end{thm}

\begin{proof}
  Por \olref[frd][acc]{thm:soundschemas}, sabemos que todas as
  instâncias de \Ax{D} e~\Ax{5} são verdadeiras em todos os modelos
  seriais e euclidianos. Portanto, basta encontrar um modelo serial e
  euclidiano que contenha um mundo no qual alguma instância de~\Ax{4}
  falhe. Considere o modelo de \olref{fig:KD5not4} e observe que
  $\mSat/{M}{\Box p \lif \Box\Box p}[w_1]$.
\end{proof}
