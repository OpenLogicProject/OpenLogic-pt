\olsection{O lema da verdade}

O modelo canônico~$\mModel{M^\Sigma}$ é definido de modo que
$\mSat{M^\Sigma}{!A}[\Delta]$ se e somente se $!A \in \Delta$. Para
!!{propositional
variable}s, a definição de $V^\Sigma$ produz isso
diretamente. Contudo, devemos verificar que a equivalência vale para
todas as !!{formula}s. Fazemos isso por indução. O passo indutivo
envolve demonstrar a equivalência para !!{formula}s que contêm
operadores proposicionais (caso em que devemos usar
\olref[ccs]{prop:ccs-properties}) e operadores modais (caso em que
invocamos os resultados de \olref[mod]{sec}).

\begin{prop}[Lema da verdade]
  \ollabel{prop:truthlemma}
  Para toda !!{formula}~$!A$, $\mSat{M^\Sigma}{!A}[\Delta]$ se e somente se
  $!A \in \Delta$.
\end{prop}

\begin{proof}
  Por indução em~$!A$.
  \begin{enumerate}
    \tagitem{prvFalse}{\indcase{!A}{\lfalse}{$\mSat/{M^\Sigma}{\lfalse}[\Delta]$
        por \olref[syn][trw]{defn:mmodels}, e $\lfalse \notin \Delta$
        por
        \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-lfalse}.}}{}

    \tagitem{prvTrue}{\indcase{!A}{\ltrue}{$\mSat{M^\Sigma}{\ltrue}[\Delta]$
        por \olref[syn][trw]{defn:mmodels}, e $\ltrue \in \Delta$ por
        \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-ltrue}.}}{}

    \item \indcase{!A}{p}{$\mSat{M^\Sigma}{p}[\Delta]$ se e somente se $\Delta \in
      V^\Sigma(p)$ por \olref[syn][trw]{defn:mmodels}. Além disso, $\Delta
      \in V^\Sigma(p)$ se e somente se $p \in \Delta$ por definição
      de~$V^\Sigma$.}

    \tagitem{prvNot}{\indcase{!A}{\lnot !B}{%
        \iftag{probNot}{Exercício.}{$\mSat{M^\Sigma}{\lnot !B}[\Delta]$
          se e somente se $\mSat/{M^\Sigma}{!B}[\Delta]$
          (\olref[syn][trw]{defn:mmodels}) se e somente se $!B \notin \Delta$ (pela
          hipótese de indução) se e somente se $\lnot !B \in \Delta$ (por
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-lnot}).}}}{}

    \tagitem{prvAnd}{\indcase{!A}{!B \land !C}{%
        \iftag{probAnd}{Exercício.}{$\mSat{M^\Sigma}{!B \land
            !C}[\Delta]$ se e somente se $\mSat{M^\Sigma}{!B}[\Delta]$ e
          $\mSat{M^\Sigma}{!C}[\Delta]$ (por
          \olref[syn][trw]{defn:mmodels}) se e somente se $!B \in \Delta$ e $!C
          \in \Delta$ (pela hipótese de indução) se e somente se $!B \land !C \in
          \Delta$ (por
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-land}).}}}{}

    \tagitem{prvOr}{\indcase{!A}{!B \lor !C}{%
        \iftag{probOr}{Exercício.}{$\mSat{M^\Sigma}{!B \lor
            !C}[\Delta]$ se e somente se $\mSat{M^\Sigma}{!B}[\Delta]$ ou
          $\mSat{M^\Sigma}{!C}[\Delta]$ (por
          \olref[syn][trw]{defn:mmodels}) se e somente se $!B \in \Delta$ ou $!C
          \in \Delta$ (pela hipótese de indução) se e somente se $!B \lor !C \in
          \Delta$ (por
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-lor}).}}}{}

    \tagitem{prvIf}{\indcase{!A}{!B \lif !C}{%
        \iftag{probIf}{Exercício.}{$\mSat{M^\Sigma}{!B \lif
            !C}[\Delta]$ se e somente se $\mSat/{M^\Sigma}{!B}[\Delta]$ ou
          $\mSat{M^\Sigma}{!C}[\Delta]$ (por
          \olref[syn][trw]{defn:mmodels}) se e somente se $!B \notin \Delta$ ou
          $!C \in \Delta$ (pela hipótese de indução) se e somente se $!B \lif !C
          \in \Delta$ (por
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-lif}).}}}{}

    \tagitem{prvIff}{\indcase{!A}{!B \liff !C}{%
        \iftag{probIff}{Exercício.}{$\mSat{M^\Sigma}{!B \liff
            !C}[\Delta]$ se e somente se ou $\mSat{M^\Sigma}{!B}[\Delta]$ e
          $\mSat{M^\Sigma}{!C}[\Delta]$ ou
          $\mSat/{M^\Sigma}{!B}[\Delta]$ e
          $\mSat/{M^\Sigma}{!C}[\Delta]$ (por
          \olref[syn][trw]{defn:mmodels}) se e somente se ou $!B \in \Delta$
          e $!C \in \Delta$ ou $!B \notin \Delta$ e $!C \notin
          \Delta$ (pela hipótese de indução) se e somente se $!B \liff !C \in
          \Delta$ (por
          \olref[ccs]{prop:ccs-properties}\olref[ccs]{prop:ccs-liff}).}}}{}

    \tagitem{prvBox}{\indcase{!A}{\Box !B}{%
        \iftag{probBox}{Exercício.}{Primeiro, suponha que
          $\mSat{M^\Sigma}{\Box !B}[\Delta]$. Por
          \olref[syn][trw]{defn:mmodels}, para todo $\Delta'$ tal
          que $R^\Sigma \Delta\Delta'$,
          $\mSat{M^\Sigma}{!B}[\Delta']$. Pela hipótese de indução, para
          todo $\Delta'$ tal que $R^\Sigma \Delta\Delta'$, $!B \in
          \Delta'$. Pela definição de $R^\Sigma$, para todo $\Delta'$
          tal que $\Box^{-1} \Delta \subseteq \Delta'$, $!B \in
          \Delta'$. Por \olref[mod]{prop:box}, $\Box !B \in
          \Delta$.

          Suponha agora $\Box !B \in \Delta$. Seja
          $\Delta' \in W^\Sigma$ tal que $R^\Sigma \Delta\Delta'$, O.s.\@, $\Box^{-1}\Delta \subseteq \Delta'$. Como
          $\Box !B \in \Delta$, $!B \in \Box^{-1} \Delta$.
          Consequentemente, $!B \in \Delta'$. Pela hipótese de indução,
          $\mSat{M^\Sigma}{!B}[\Delta']$. Como $\Delta'$ é arbitrário
          com $R^\Sigma \Delta\Delta'$, para todo
          $\Delta' \in W^\Sigma$ tal que $R^\Sigma \Delta\Delta'$,
          $\mSat{M^\Sigma}{!B}[\Delta']$. Por
          \olref[syn][trw]{defn:mmodels}, $\mSat{M^\Sigma}{\Box
            !B}[\Delta]$.}}}{}

    \tagitem{prvDiamond}{\indcase{!A}{\Diamond !B}{%
        \iftag{probDiamond}{Exercício.}{Primeiro, suponha que
          $\mSat{M^\Sigma}{\Diamond !B}[\Delta]$. Por
          \olref[syn][trw]{defn:mmodels}, para algum $\Delta'$ tal que
          $R^\Sigma \Delta\Delta'$, $\mSat{M^\Sigma}{!B}[\Delta']$.
          Pela hipótese de indução, para algum $\Delta'$ tal que
          $R^\Sigma \Delta\Delta'$, $!B \in \Delta'$. Pela definição de
          $R^\Sigma$,\iftag{prvBox}{ para algum $\Delta'$ tal que
            $\Box^{-1} \Delta \subseteq \Delta'$, $!B \in \Delta'$.
            Por \olref[mod]{prop:diamond},}{} para algum $\Delta'$ tal
          que $\Diamond \Delta' \subseteq \Delta$, $!B \in
          \Delta'$. Como $!B \in \Delta'$, $\Diamond !B \in \Diamond
          \Delta'$, logo $\Diamond !B \in \Delta$.

          Suponha agora $\Diamond !B \in \Delta$. Por
          \olref[mod]{prop:diamond}, existe um conjunto completo
          $\Sigma$-consistente $\Delta' \in W^\Sigma$ tal que
          $\Diamond \Delta' \subseteq \Delta$ e $!B \in
          \Delta'$.\iftag{prvBox}{ Por \olref[mod]{lem:box-iff-diamond},
            existe um $\Delta' \in W^\Sigma$ tal que $\Box^{-1}
            \Delta \subseteq \Delta'$, e $!B \in \Delta'$.}{} Pela
          definição de $R^\Sigma$, $R^\Sigma \Delta\Delta'$; logo,
          existe um $\Delta' \in W^\Sigma$ tal que $R^\Sigma
          \Delta\Delta'$ e $!B \in \Delta'$. Por
          \olref[syn][trw]{defn:mmodels}, $\mSat{M^\Sigma}{\Diamond
            !B}[\Delta]$.}}}{}
  \end{enumerate}
\end{proof}

\begin{probtag}{probFalse,probTrue,probNot,proband,probOr,probIf,probIff,probBox,probDiamond}
  Complete a demonstração de \olref[nml][com][tru]{prop:truthlemma}.
\end{probtag}
