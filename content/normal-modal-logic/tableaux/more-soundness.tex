\olsection{Correção das regras adicionais}

Dizemos que uma regra é correta para uma classe de modelos se, sempre que
um ramo de !!a{tableau} é satisfatível em um modelo dessa classe, o ramo
resultante da aplicação da regra também é satisfatível em um modelo dessa
classe.

\begin{prop}
  \ollabel{prop:soundness-T} 
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ e }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{é correta}{são corretas}
  para modelos reflexivos.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{O ramo é expandido pela aplicação de T$\Box$ a
  $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$:
  \iftag{probBox}{Exercício.}{Isso produz uma nova !!{signed
      formula}~$\sFmla{\True}{!B}[\sigma]$ no ramo. Suponha
    $\mSat{M}{\Gamma}[f]$, em particular, $\mSat{M}{\Box
      !B}[f(\sigma)]$. Como $R$ é reflexiva, sabemos que
    $Rf(\sigma)f(\sigma)$. Logo, $\mSat{M}{!B}[f(\sigma)]$, O.s.\@,
    $\mModel{M}, f$ satisfaz $\sFmla{\True}{!B}[\sigma]$.}}{}
\tagitem{prvDiamond}{O ramo é expandido pela aplicação de T$\Diamond$ a
  $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$:
  \iftag{probDiamond}{Exercício.}{Isso produz uma nova !!{signed
      formula}~$\sFmla{\False}{!B}[\sigma]$ no ramo. Suponha
    $\mSat{M}{\Gamma}[f]$, em particular, $\mSat/{M}{\Diamond
      !B}[f(\sigma)]$. Como $R$ é reflexiva, sabemos que
    $Rf(\sigma)f(\sigma)$. Logo, $\mSat/{M}{!B}[f(\sigma)]$, O.s.\@,
    $\mModel{M}, f$ satisfaz $\sFmla{\False}{!B}[\sigma]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  Complete a demonstração de \olref[nml][tab][msn]{prop:soundness-T}
\end{prob}
\tagendprob

\begin{prop}
  \ollabel{prop:soundness-D} 
  \iftag{prvBox}{\Ax{D}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ e }%
  \iftag{prvDiamond}{\Ax{D}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{é correta}{são corretas}
  para modelos seriais.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{O ramo é expandido pela aplicação de D$\Box$ a
  $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$:
  \iftag{probBox}{Exercício.}{Isso produz uma nova !!{signed
      formula}~$\sFmla{\True}{\Diamond !B}[\sigma]$ no ramo. Suponha
    $\mSat{M}{\Gamma}[f]$, em particular, $\mSat{M}{\Box
      !B}[f(\sigma)]$. Como $R$ é serial, há um $w \in W$ tal que
    $Rf(\sigma)w$. Então $\mSat{M}{!B}[w]$ e, portanto,
    $\mSat{M}{\Diamond !B}[f(\sigma)]$. Logo, $\mModel{M}, f$ satisfaz
    $\sFmla{\True}{\Diamond!B}[\sigma]$.}}{}
\tagitem{prvDiamond}{O ramo é expandido pela aplicação de D$\Diamond$ a
  $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$:
  \iftag{probDiamond}{Exercício.}{Isso produz uma nova !!{signed
      formula}~$\sFmla{\False}{\Box !B}[\sigma]$ no ramo. Suponha
    $\mSat{M}{\Gamma}[f]$, em particular,
    $\mSat/{M}{\Diamond !B}[f(\sigma)]$. Como $R$ é serial, há um
    $w \in W$ tal que $Rf(\sigma)w$. Então $\mSat/{M}{!B}[w]$ e,
    portanto, $\mSat/{M}{\Box !B}[f(\sigma)]$. Logo, $\mModel{M}, f$
    satisfaz $\sFmla{\False}{\Box!B}[\sigma]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  Complete a demonstração de \olref[nml][tab][msn]{prop:soundness-D}
\end{prob}
\tagendprob

\begin{prop}
  \ollabel{prop:soundness-B} 
  \iftag{prvBox}{\Ax{B}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ e }%
  \iftag{prvDiamond}{\Ax{B}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{é correta}{são corretas}
  para modelos simétricos.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{O ramo é expandido pela aplicação de B$\Box$ a
  $\sFmla{\True}{\Box !B}[\sigma.n] \in \Gamma$:
  \iftag{probBox}{Exercício.}{Isso produz uma nova !!{signed
      formula}~$\sFmla{\True}{!B}[\sigma]$ no ramo. Suponha
    $\mSat{M}{\Gamma}[f]$, em particular, $\mSat{M}{\Box
      !B}[f(\sigma.n)]$. Como $f$ é uma interpretação dos prefixos do
    ramo em~$\mModel{M}$, sabemos que $Rf(\sigma)f(\sigma.n)$. Como $R$
    é simétrica, $Rf(\sigma.n)f(\sigma)$. Como
    $\mSat{M}{\Box !B}[f(\sigma.n)]$, $\mSat{M}{!B}[f(\sigma)]$. Logo,
    $\mModel{M}, f$ satisfaz $\sFmla{\True}{!B}[\sigma]$.}}{}
\tagitem{prvDiamond}{O ramo é expandido pela aplicação de B$\Diamond$ a
  $\sFmla{\False}{\Diamond !B}[\sigma.n] \in \Gamma$:
  \iftag{probDiamond}{Exercício.}{Isso produz uma nova !!{signed
      formula}~$\sFmla{\False}{!B}[\sigma]$ no ramo. Suponha
    $\mSat{M}{\Gamma}[f]$, em particular, $\mSat/{M}{\Diamond
      !B}[f(\sigma.n)]$. Como $f$ é uma interpretação dos prefixos do
    ramo em~$\mModel{M}$, sabemos que $Rf(\sigma)f(\sigma.n)$. Como $R$
    é simétrica, $Rf(\sigma.n)f(\sigma)$. Como
    $\mSat/{M}{\Diamond !B}[f(\sigma.n)]$, $\mSat/{M}{!B}[f(\sigma)]$.
    Logo, $\mModel{M}, f$ satisfaz $\sFmla{\False}{!B}[\sigma]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  Complete a demonstração de \olref[nml][tab][msn]{prop:soundness-B}
\end{prob}
\tagendprob
