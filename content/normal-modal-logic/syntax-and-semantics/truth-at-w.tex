\olsection{Verdade em um mundo}

Todo modelo modal determina quais !!{formula}s modais contam como
verdadeiras em quais de seus mundos. A relação ``o modelo $\mModel{M}$
torna a !!{formula}~$!A$ verdadeira no mundo~$w$'' é a noção fundamental
da semântica relacional. Essa relação é definida indutivamente e coincide
com a caracterização usual por tabelas de verdade para os operadores não
modais.

\begin{defn}\ollabel{defn:mmodels}
  A \emph{verdade de !!a{formula}~$!A$ em~$w$} em um~$\mModel M$,
  simbolicamente $\mSat{M}{!A}[w]$, é definida indutivamente como segue:
  \begin{enumerate}
  \tagitem{prvFalse}{\indcase{!A}{\lfalse}{Nunca $\mSat{M}{\lfalse}[w]$}.}{}
  \tagitem{prvTrue}{\indcase{!A}{\ltrue}{Sempre $\mSat{M}{\ltrue}[w]$}.}{}
  \item $\mSat{M}{p}[w]$ sse $w \in V(p)$.
  \tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$ sse
    $\mSat/{M}{!B}[w]$}.}{}
  \tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{$\mSat{M}{\indfrm}[w]$ sse
    $\mSat{M}{!B}[w]$ e $\mSat{M}{!C}[w]$}.}{}
  \tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{$\mSat{M}{\indfrm}[w]$ sse
    $\mSat{M}{!B}[w]$ ou $\mSat{M}{!C}[w]$} (ou ambos).}{}
  \tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{$\mSat{M}{\indfrm}[w]$ sse
    $\mSat/{M}{!B}[w]$ ou $\mSat{M}{!C}[w]$}.}{}
  \tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{$\mSat{M}{\indfrm}[w]$ sse
    ou valem ambos $\mSat{M}{!B}[w]$ e $\mSat{M}{!C}[w]$, ou não vale
    nenhum dos dois, $\mSat{M}{!B}[w]$ e $\mSat{M}{!C}[w]$}.}{}
  \item\ollabel{defn:sub:mmodels-box}
    \indcase{!A}{\Box !B}{$\mSat{M}{\indfrm}[w]$ sse
    $\mSat{M}{!B}[w']$ para todo $w' \in W$ tal que $Rww'$.}
  \item\ollabel{defn:sub:mmodels-diamond}
    \indcase{!A}{\Diamond !B}{$\mSat{M}{\indfrm}[w]$ sse
    $\mSat{M}{!B}[w']$ para pelo menos um $w' \in W$ tal que $Rww'$.}
  \end{enumerate}
\end{defn}

Observe que, pela cláusula~\olref{defn:sub:mmodels-box}, !!a{formula}
~$\Box !B$ é verdadeira em~$w$ sempre que não há~$w'$ tal que $Rww'$.
Nesse caso, $\Box !B$ é verdadeira \emph{vacuamente} em~$w$. Além disso,
$\Box !B$ pode ser satisfeita em~$w$ mesmo que $!B$ não o seja. A verdade
de~$!B$ em~$w$ não garante a verdade de~$\Diamond !B$ em~$w$. Isso vale,
porém, se $Rww$, por exemplo, se $R$ é reflexiva. Se não há $w'$ tal que
$Rww'$, então $\mSat/{M}{\Diamond !A}[w]$, para qualquer~$!A$.

\begin{prob}
  Considere o modelo de \olref[nml][syn][rel]{fig:simple}. Quais das
  afirmações a seguir valem?
    \begin{enumerate}
    \item $\mSat{M}{q}[w_1]$;
    \item $\mSat{M}{\lnot q}[w_3]$;
    \item $\mSat{M}{p \lor q}[w_1]$;
    \item $\mSat{M}{\Box (p \lor q)}[w_1]$;
    \item $\mSat{M}{\Box q}[w_3]$;
    \item $\mSat{M}{\Box \bot}[w_3]$;
    \item $\mSat{M}{\Diamond q}[w_1]$;

    \end{enumerate}
\end{prob}

% Prove duality of \Box and \Diamond
\begin{tagblock}{prvDiamond}
\begin{prop}\ollabel{prop:dual}
  \begin{enumerate}
  \item $\mSat{M}{\Box !A}[w]$ iff $\mSat{M}{\lnot\Diamond\lnot !A}[w]$.
  \item $\mSat{M}{\Diamond !A}[w]$ iff $\mSat{M}{\lnot\Box\lnot !A}[w]$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ sse $\mSat/{M}{\Diamond\lnot
      !A}[w]$, pela definição de $\mSat{M}{}[w]$. $\mSat{M}{\Diamond\lnot
      !A}[w]$ sse, para algum $w'$ tal que $Rww'$, $\mSat{M}{\lnot
      !A}[w']$. Logo, $\mSat/{M}{\Diamond\lnot !A}[w]$ sse, para todo
    $w'$ tal que $Rww'$, $\mSat/{M}{\lnot !A}[w']$. Também temos
    $\mSat/{M}{\lnot !A}[w']$ sse $\mSat{M}{!A}[w']$. Em conjunto,
    obtemos que $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ sse, para todo $w'$
    tal que $Rww'$, $\mSat{M}{!A}[w']$. Novamente pela definição de
    $\mSat{M}{}[w]$, isso ocorre sse $\mSat{M}{\Box !A}[w]$.
    \item \iftag{probDiamond}{Exercício.}{$\mSat{M}{\lnot\Box\lnot !A}[w]$ sse
      $\mSat/{M}{\Box\lnot !A}$. $\mSat{M}{\Box\lnot !A}[w]$ sse, para todo
      $w'$ tal que $Rww'$, $\mSat{M}{\lnot !A}[w']$. Logo,
      $\mSat/{M}{\Box\lnot !A}[w]$ sse, para algum $w'$ tal que $Rww'$,
      $\mSat/{M}{\lnot !A}[w']$. Também temos $\mSat/{M}{\lnot
        !A}[w']$ sse $\mSat{M}{!A}[w']$. Em conjunto, obtemos que
      $\mSat{M}{\lnot\Box\lnot !A}[w]$ sse, para algum~$w'$ tal que
      $Rww'$, $\mSat{M}{!A}[w']$. Novamente pela definição de
      $\mSat{M}{}[w]$, isso ocorre sse $\mSat{M}{\Diamond
        !A}[w]$.}
  \end{enumerate}
\end{proof}

\begin{probtag}{probDiamond}
  Complete a demonstração de \olref[nml][syn][trw]{prop:dual}.
\end{probtag}
\end{tagblock}

\begin{prob}
  Seja $\mModel{M} = \tuple{W, R, V}$ um modelo e suponha que $w_1, w_2
  \in W$ sejam tais que:
  \begin{enumerate}
  \item $w_1 \in V(p)$ se, e somente se, $w_2 \in V(p)$ (para toda !!{propositional variable} $p$); e
  \item para todo $w \in W$: $Rw_1w$ se, e somente se, $Rw_2w$.
  \end{enumerate}
  Usando indução sobre !!{formula}s, mostre que, para todas as !!{formula}s $!A$:
  $\mSat{M}{!A}[w_1]$ se, e somente se, $\mSat{M}{!A}[w_2]$.
\end{prob}

\begin{probtag}{prvDiamond}
  Seja $\mModel{M} = \tuple{W, R, V}$. Mostre que
  $\mSat{M}{\lnot\Diamond !A}[w]$ se, e somente se,
  $\mSat{M}{\Box\lnot!A}[w]$.
\end{probtag}
