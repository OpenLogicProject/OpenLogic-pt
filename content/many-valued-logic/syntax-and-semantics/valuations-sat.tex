\olsection{\usetoken{P}{valuation} e satisfação}

\begin{defn}[!!^{valuation}s] 
Seja $V$ um conjunto de valores de verdade. Uma \emph{!!{valuation}} para
$\Lang{L}$ em~$V$ é uma
função~$\pAssign{v}$ que atribui !!a{element} de~$V$ às
!!{propositional variable}s da linguagem, isto é, $\pAssign{v} \colon
\PVar \to V$.
\end{defn}

\begin{defn}\ollabel{defn:pValue}
  Dada !!a{valuation}~$\pAssign{v}$ no conjunto de valores de verdade~$V$ de uma
  lógica multivalorada~$\Log L$, definimos indutivamente a função de avaliação
  $\pValue{v} \colon \Frm[L] \to V$ por:
  \begin{enumerate}
    \item $\pValue{v}(\Obj p_n) = \pAssign{v}(\Obj p_n)$; 
    \item Se $\star$ é um conectivo $0$-ário, então $\pValue{v}(\star)  = \tf{\star}[\Log L]$;
    \item Se $\star$ é um conectivo $n$-ário, então 
    \[
      \pValue{v}(\star(!A_1, \dots, !A_n)) = \tf{\star}[\Log L]
      (\pValue{v}(!A_1), \dots, \pValue{v}(!A_n)).
    \]
  \end{enumerate}
\end{defn}

\begin{defn}[Satisfação]
\ollabel{defn:satisfaction} A !!{formula}~$!A$ é \emph{satisfeita}
  por !!a{valuation}~$\pAssign{v}$, $\pSat{v}{!A}[\Log L]$, se, e somente se,
  $\pValue{v}(!A)[\Log L] \in V^+$, onde $V^+$ é o conjunto dos
  valores de verdade designados de~$\Log L$. 
  
  Escrevemos $\pSat/{v}{!A}[\Log L]$ para dizer ``não $\pSat{v}{!A}[\Log
  L]$.'' Se $\Gamma$ é um conjunto de !!{formula}s, $\pSat{v}{\Gamma}[\Log
  L]$ se, e somente se, $\pSat{v}{!A}[\Log L]$ para todo~$!A \in \Gamma$.
\end{defn}
