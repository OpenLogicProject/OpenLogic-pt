\olsection{Noções semânticas}

\begin{defn}
  Dizemos que $!A$ é \emph{verdadeira no modelo
  $\mModel{M} = \tuple{W,R,V}$,} $\mSat{M}{!A}$, sse
  $\mSat{M}{!A}[w]$ para todo $w \in W$. $!A$ é \emph{válida},
  $\Entails !A$, sse é verdadeira em todos os modelos. Dizemos que um
  conjunto de !!{formula}s~$\Gamma$ \emph{implica}~$!A$, $\Gamma
  \Entails !A$, sse, para todo modelo~$\mModel{M}$ e todo~$w$ tais
  que $\mSat{M}{\Gamma}[w]$, $\mSat{M}{!A}[w]$.
\end{defn}

\begin{prop}\ollabel{prop:sat-entails}
  \begin{enumerate}
  \item\ollabel{prop:sat-entails1} Se $\mSat{M}{\Gamma}[w]$ e
    $\Gamma \Entails !A$, então $\mSat{M}{!A}[w]$.
  \item\ollabel{prop:sat-entails2} Se $\mSat{M}{\Gamma}$ e $\Gamma
    \Entails !A$, então $\mSat{M}{!A}$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item Suponha $\mSat{M}{\Gamma}$. Como $\Gamma \Entails !A$,
    sabemos que, se $\mSat{M}{\Gamma}[w]$, então
    $\mSat{M}{!A}[w]$. Como $\mSat{M}{\Gamma}[u]$ para todo
    $u \in W$, $\mSat{M}{\Gamma}[w]$. Logo $\mSat{M}{!A}[w]$.
  \item Segue imediatamente de \olref{prop:sat-entails1}.
  \end{enumerate}
\end{proof}


\begin{defn}\ollabel{defn:restrict}
  Suponha que $\mModel{M}$ seja um modelo relacional e que $w \in W$.
  A \emph{restrição} $\mModel{M}_w=\tuple{W_w, R_w, V_w}$
  de~$\mModel{M}$ a~$w$ é dada por:
  \begin{align*}
    W_w & = \Setabs{u \in W}{Rwu},\\
    R_w  & = R \cap (W_w)^2, \text{ e}\\
    V_w(p) & = V(p) \cap W_w.
  \end{align*}
\end{defn}

\begin{prop}\ollabel{prop:restrict}
  $\mSat{M}{!A}[w]$ sse $\mSat{M_w}{!A}$.
\end{prop}

\begin{prob}
  Prove \olref[int][sem][sem]{prop:restrict}. 
\end{prob}

\begin{prop}
  Suponha que, para todo modelo~$\mModel{M}$ tal que
  $\mSat{M}{\Gamma}$, tenhamos $\mSat{M}{!A}$. Então
  $\Gamma \Entails !A$.
\end{prop}

\begin{proof}
  Suponha $\mSat{M}{\Gamma}[w]$. Pela \olref{prop:restrict} aplicada
  a todo~$!B \in \Gamma$, temos $\mSat{M_w}{\Gamma}$. Pela hipótese,
  temos $\mSat{M_w}{!A}$. Aplicando novamente
  \olref{prop:restrict}, obtemos $\mSat{M}{!A}[w]$.
\end{proof}
