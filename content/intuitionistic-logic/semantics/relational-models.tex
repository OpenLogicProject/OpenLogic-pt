\olsection{\usetoken{P}{relational model}}

Para oferecer uma semântica precisa da lógica proposicional
intuicionista, precisamos definir o que conta como um modelo em relação
ao qual possamos avaliar !!{formula}s. Com base nessa definição,
também é possível definir noções semânticas como validade e
consequência. Uma dessas semânticas é dada por !!{relational model}s.

\begin{defn}
  !!^a{relational model} para a lógica proposicional intuicionista é
  uma tripla $\mModel{M} = \tuple{W, R, V}$, em que
  \begin{enumerate}
  \item $W$ é um conjunto não vazio;
  \item $R$ é uma ordem parcial (isto é, uma relação binária reflexiva,
    antissimétrica e transitiva) sobre~$W$; e
  \item $V$ é uma função que atribui a cada !!{propositional
    variable}~$p$ um subconjunto de~$W$, de modo que
  \item $V$ é monotônica em relação a $R$, isto é, se $w
    \in V(p)$ e $Rww'$, então $w' \in V(p)$.
  \end{enumerate}
\end{defn}

\begin{defn}\ollabel{defn:true-at-w}
  Definimos a noção de $!A$ \emph{ser verdadeira em $w$ em
    $\mModel{M}$}, $\mSat{M}{!A}[w]$, indutivamente da seguinte maneira:
  \begin{enumerate}
  \item \indcase{!A}{p}{$\mSat{M}{\indfrm}[w]$ sse $w \in V(p)$.}
  \item \indcase{!A}{\lfalse}{não $\mSat{M}{\indfrm}[w]$}.
  \item \indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$ sse não existe
    $w'$ tal que $Rww'$ e $\mSat{M}{!B}[w']$}.
  \item \indcase{!A}{!B \land !C}{$\mSat{M}{\indfrm}[w]$ sse
    $\mSat{M}{!B}[w]$ e $\mSat{M}{!C}[w]$}.
  \item \indcase{!A}{!B \lor !C}{$\mSat{M}{\indfrm}[w]$ sse
    $\mSat{M}{!B}[w]$ ou $\mSat{M}{!C}[w]$ (ou ambos)}.
  \item \indcase{!A}{!B \lif !C}{$\mSat{M}{\indfrm}[w]$ sse, para todo
    $w'$ tal que $Rww'$, não $\mSat{M}{!B}[w']$ ou
    $\mSat{M}{!C}[w']$} (ou ambos).
  \end{enumerate}
  Escrevemos $\mSat/{M}{!A}[w]$ se não $\mSat{M}{!A}[w]$. Se $\Gamma$ é
  um conjunto de !!{formula}s, $\mSat{M}{\Gamma}[w]$ significa
  $\mSat{M}{!B}[w]$ para todo $!B \in \Gamma$.
\end{defn}

\begin{prob}
  Mostre que, de acordo com \olref[int][sem][rel]{defn:true-at-w},
  $\mSat{M}{\lnot !A}[w]$ sse $\mSat{M}{!A \lif \lfalse}[w]$.
\end{prob}

\begin{prop}\ollabel{prop:true-monotonic}
  A verdade nos mundos é monotônica em relação a~$R$, isto é, se
  $\mSat{M}{!A}[w]$ e $Rww'$, então $\mSat{M}{!A}[w']$.
\end{prop}

\begin{proof}
  Exercício.
\end{proof}

\begin{prob}
  Prove \olref[int][sem][rel]{prop:true-monotonic}.
\end{prob}
