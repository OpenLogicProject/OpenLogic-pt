\olsection{Minimização}

As funções recursivas gerais são aquelas que podem ser obtidas a partir
das funções básicas $\Zero$, $\Succ$ e $\Proj{n}{i}$ por composição,
recursão primitiva e minimização regular. Para mostrar que todas as
funções recursivas gerais são !!{lambda definable}, precisamos mostrar
que qualquer função definida por minimização regular a partir de uma função
!!a{lambda definable} é, ela própria, !!{lambda definable}.

\begin{lem}
  \ollabel{lem:min} Se $f(x_1, \dots, x_k, y)$ é regular e
  !!{lambda definable}, então $h$,~definida por
  \[
  h(x_1, \dots, x_k) = \umin{y}{f(x_1,\dots,x_k, y) = 0}
  \]
  também é !!{lambda definable}.
\end{lem}

\begin{proof}
  Suponha que o termo lambda~$F$ defina-$\lambda$ a função regular
  $f(\vec x, y)$. Para !!{lambda define}~$h$, usamos uma função de busca
  e um combinador de ponto fixo:
  \begin{align*}
    \fn{Search} & \ident \lambd[g][\lambd[f\,\vec{x}\,y][
        \fn{IsZero} (f\, \vec{x}\, y)\, y\, (g\, f\, \vec{x} (\fn{Succ}\, y)]]]\\
    H & \ident \lambd[\vec x][(Y \, \fn{Search}) F\, \vec{x}\, \num{0}],
  \end{align*}
  onde $Y$ é qualquer combinador de ponto fixo. Informalmente,
  $\fn{Search}$ é uma função autorreferencial: começando com~$y$,
  testa se $f\, \vec x\, y$ é zero; se for, devolve~$y$; caso contrário, chama
  a si mesma com $\fn{Succ}\, y$. Assim, $(Y \, \fn{Search}) F
  \num{n_1}\dots\num{n_k}\,\num{0}$ devolve o menor~$m$ para o qual $f(n_1,
  \dots, n_k, m) = 0$.
  
  Mais especificamente, observe que
  \begin{align*}
    (Y \, \fn{Search}) F \num{n_1}
    \dots\num{n_k}\, \num{m} & \red \num{m}
    \intertext{se $f(n_1, \dots,
      n_k, m) = 0$, ou}
    & \red (Y \, \fn{Search}) F\, \num{n_1} \dots\num{n_k}\, \num{m+1}
    \intertext{caso contrário. Como $f$ é regular, $f(n_1, \dots, n_k, y)
      = 0$ para algum $y$ e, portanto,}
    (Y \, \fn{Search}) F \num{n_1} \dots\num{n_k}\,\num{0}
    & \red \num{h(n_1, \dots, n_k)}.
    \end{align*}
\end{proof}


\begin{prop}
  Toda função recursiva geral é !!{lambda definable}.
\end{prop}

\begin{proof}
 Por \olref[prf]{lem:basic}, todas as funções básicas são
 !!{lambda definable} e, por \olref[prf]{lem:comp},
 \olref[prf]{lem:prim} e \olref{lem:min}, as funções !!{lambda definable}
 são fechadas sob composição, recursão primitiva e
 minimização regular.
\end{proof}
